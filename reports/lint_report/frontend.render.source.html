185 def _draw_pacman(self, surface: pygame.Surface, pacman: PacMan) -> None:
186 cx = int(pacman.position.x)
187 cy = int(pacman.position.y) + self.HUD_HEIGHT
188 radius = self.tile // 2 - 2
189 angle = pacman.mouth_angle
190 rot = pacman.rotation_deg
191  
192 # Draw filled arc (pie slice cut out for mouth)
193 start_deg = rot + angle
194 end_deg = rot + 360 - angle
195 self._draw_pacman_arc(surface, cx, cy, radius, start_deg, end_deg, YELLOW)
196 # Eye
197 eye_angle = math.radians(rot + 70)
198 ex = cx + int(math.cos(eye_angle) * radius * 0.5)
199 ey = cy - int(math.sin(eye_angle) * radius * 0.5)
200 pygame.draw.circle(surface, BLACK, (ex, ey), 2)

202 @staticmethod
203 def _draw_pacman_arc(
204 surface: pygame.Surface,
205 cx: int, cy: int, radius: int,
206 start_deg: float, end_deg: float,
207 color: tuple,
208 ) -> None:
209 """Draw a filled pie/arc shape."""
210 points = [(cx, cy)]
211 for deg in range(int(start_deg), int(end_deg) + 1, 3):
212 rad = math.radians(deg)
213 x = cx + int(math.cos(rad) * radius)
214 y = cy - int(math.sin(rad) * radius)
215 points.append((x, y))
216 if len(points) >= 3:
217 pygame.draw.polygon(surface, color, points)

222 cx = int(ghost.position.x)
223 cy = int(ghost.position.y) + self.HUD_HEIGHT
224 r = self.tile // 2 - 2
225  
226 if ghost.mode == GhostMode.FRIGHTENED:

233 else:
234 body_color = ghost.color
235  
236 # Body: top semicircle + rectangle bottom with wavy skirt
237 body_rect = pygame.Rect(cx - r, cy - r, r * 2, r * 2)
238 pygame.draw.ellipse(surface, body_color, body_rect)
239 lower_rect = pygame.Rect(cx - r, cy, r * 2, r)
240 pygame.draw.rect(surface, body_color, lower_rect)
241  
242 # Wavy skirt (3 bumps)
243 bump_r = r // 3
244 for i in range(3):
245 bx = cx - r + bump_r + i * bump_r * 2
246 by = cy + r
247 pygame.draw.circle(surface, body_color, (bx, by), bump_r)
248  
249 # Eyes
250 self._draw_ghost_eyes(surface, cx, cy, r)
251  
252 # Frightened face
253 if ghost.mode == GhostMode.FRIGHTENED:
254 # Squiggly mouth

256 h = 3 if (i // 4) % 2 == 0 else -3
